# ==============================================================================
# CoopServe Platform — Software Requirements & Dependency Manifest
# ==============================================================================
# NOTE: CoopServe is a JavaScript / Node.js full-stack platform (not Python).
# Dependencies are managed via npm / package.json in backend/ and frontend/.
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. System & Runtime Prerequisites
# ------------------------------------------------------------------------------
Node.js >= 18.0.0 (Recommended: Node.js 20 LTS)
npm >= 9.0.0 (comes bundled with Node.js)
MongoDB >= 6.0 (Local instance or MongoDB Atlas URI)

# ------------------------------------------------------------------------------
# 2. Backend Dependencies (from backend/package.json)
# ------------------------------------------------------------------------------
# Install command: cd backend && npm install

cors==2.8.5                # Cross-Origin Resource Sharing middleware
dotenv==16.4.7             # Environment configuration loader
express==4.21.2            # REST API web framework
jsonwebtoken==9.0.2        # JWT token generation & verification
mongoose==8.9.5            # MongoDB object modeling
morgan==1.10.0             # HTTP request logger
razorpay==2.9.6            # Razorpay Payment Gateway SDK
socket.io==4.8.1           # Real-time WebSockets engine
